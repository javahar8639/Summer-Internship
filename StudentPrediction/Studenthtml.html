# Machine Learning Concepts
# Drop rows with missing target values
sp = sp.dropna(subset=["Passed"])

sp["Attendance Rate"] = sp["Attendance Rate"].fillna(sp["Attendance Rate"].median())
sp["Previous Grades"] = sp["Previous Grades"].fillna(sp["Previous Grades"].median())

# Fill missing categorical values with most frequent

    sp["Participation in Extracurricular Activities"].mode()[0]
